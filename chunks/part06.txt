<script>
(function(){
  const deck=document.getElementById('deck');
  const slides=[...document.querySelectorAll('.slide')];
  const counter=document.getElementById('counter');
  let index=0;
  const qs=new URLSearchParams(location.search);
  const start=parseInt(qs.get('slide')||'1',10);
  if(start>=1&&start<=slides.length) index=start-1;
  function fit(){
    const s=Math.min(window.innerWidth/1600,window.innerHeight/900);
    deck.style.transform=`translate(-50%,-50%) scale(${s})`;
  }
  function show(i){
    index=Math.max(0,Math.min(slides.length-1,i));
    slides.forEach((s,n)=>s.classList.toggle('active',n===index));
    counter.textContent=`${index+1} / ${slides.length}`;
    document.title=`${slides[index].dataset.title} | Presentación proveedores`;
  }
  function next(){show(index+1)}
  function prev(){show(index-1)}
  document.getElementById('next').onclick=next;
  document.getElementById('prev').onclick=prev;
  document.getElementById('fullscreen').onclick=()=>{
    if(!document.fullscreenElement){
      const fn=document.documentElement.requestFullscreen||document.documentElement.webkitRequestFullscreen;
      if(fn) fn.call(document.documentElement);
    }else{
      const fn=document.exitFullscreen||document.webkitExitFullscreen;
      if(fn) fn.call(document);
    }
  };
  document.getElementById('print').onclick=()=>window.print();
  addEventListener('resize',fit);
  addEventListener('keydown',e=>{
    if(['ArrowRight','PageDown',' '].includes(e.key)){e.preventDefault();next();}
    if(['ArrowLeft','PageUp'].includes(e.key)){e.preventDefault();prev();}
    if(e.key==='Home'){e.preventDefault();show(0);}
    if(e.key==='End'){e.preventDefault();show(slides.length-1);}
    if(e.key==='f'||e.key==='F'){e.preventDefault();document.getElementById('fullscreen').click();}
  });
  fit(); show(index);
})();
</script>
